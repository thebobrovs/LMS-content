@jax.jit                  # XLA compiles + fuses this graph
def layer(x, w, b, scale):
    y = x @ w            # MatMul  — the heavy op
    y = y + b            # + bias  ┐
    y = jax.nn.gelu(y)     # GELU    │ fused into the
    y = y * scale        # × scale ┘ matmul's epilogue
    return y
